// =====================================================
// USER & ACCESS MANAGEMENT
// =====================================================

// Stores system roles such as Student, Staff, Admin
Table roles {
  role_id bigint [pk, increment]
  role_name varchar(25) [not null, unique, note: 'Student, Staff, Admin']
}

// Stores login and account information
Table users {
  user_id bigint [pk, increment]
  username varchar(9) [not null, unique, note: 'Student ID or Staff ID']
  email varchar(254) [not null, unique]
  password_hash varchar(255) [not null]
  role_id bigint [not null, ref: > roles.role_id]
  account_status_id bigint [not null, ref: > account_statuses.account_status_id, note: "On delete: restrict, default: 1"]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores the account statuses that are available
Table account_statuses {
  account_status_id bigint [pk, increment]
  account_status varchar(20) [not null, default: 'active', note: "active, disabled"]
  created_at datetime [not null]
  updated_at datetime [not null]
}


// =====================================================
// STUDENT PROFILE
// =====================================================

// Stores core student profile information
Table student_profiles {
  profile_id bigint [pk, increment]
  user_id bigint [not null, unique, ref: - users.user_id, note: 'On delete: cascade']
  first_name varchar(50)
  preferred_name varchar(50) [note: 'Optional display name']
  last_name varchar(50) [not null]
  degree_title varchar(40)
  specialisation varchar(60)
  personal_intro text
  upcoming_actions text [note: 'Future personal development actions']
  profile_image_url varchar(255)
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Log student actions
Table student_actions {
  student_action_id bigint [pk, increment]
  student_profile bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  action varchar(100) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Centralized link storage instead of mixing fixed URL columns and extra links
Table student_links {
  link_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  link_label varchar(100) [note: 'Display label shown to users']
  link_url varchar(500) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Separated from student_profiles because elevator pitch belongs more naturally to networking
Table elevator_pitches {
  pitch_id bigint [pk, increment]
  profile_id bigint [not null, unique, ref: - student_profiles.profile_id, note: 'On delete: cascade']
  pitch_text text
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table achievement_certs {
  achievement_cert_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  title varchar(40) [not null]
  body text [note: 'Upload awards, scholarships, or other certificates that demonstrate your academic or extracurricular success.']
  file_path varchar(255) [note: 'if applicable']
  issued_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table attainment_certs {
  attainment_cert_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  title varchar(40) [not null]
  body text [note: 'Include training credentials such as White Card, First Aid, WHS training, or other job-related certifications.']
  file_path varchar(255) [note: 'if applicable']
  issued_date date
  expiry_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table portfolio_notes {
  portfolio_note_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  title varchar(40) [not null]
  body text [note: 'Add links to relevant portfolio items, project work, or documentation of achievements.']
  file_path varchar(255) [note: 'if applicable']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table job_resources {
  job_resource_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  title varchar(40) [not null]
  body text [note: 'Store responses to selection criteria, interview practice videos, or other supporting materials for job applications.']
  file_path varchar(255) [note: 'if applicable']
  created_at datetime [not null]
  updated_at datetime [not null]
}


// =====================================================
// CAREER DEVELOPMENT PLAN (CDP)
// =====================================================

// Each student can have one CDP per year
Table career_development_plans {
  plan_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  plan_year int [not null]
  professional_interests text
  employers_of_interest text
  networking_plan text
  personal_values text
  extracurriculars text
  development_focus text
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (profile_id, plan_year) [unique]
  }
}


// =====================================================
// SMART GOALS
// =====================================================

// Stores SMART goals under a specific career plan
Table smart_goals {
  goal_id bigint [pk, increment]
  plan_id bigint [not null, ref: > career_development_plans.plan_id, note: 'On delete: cascade']
  goal_description text [not null]
  timeline text
  progress_notes text
  learnings text
  start_date date
  end_date date
  completion_date date
  completion_notes text
  goal_status_id bigint [not null, ref: > goal_statuses.goal_status_id, note: 'On delete: restrict']
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Store the possible selections for goal_statuus
Table goal_statuses {
  goal_status_id bigint [pk, increment]
  status varchar(25) [note: "planned, in_progress, completed", not null, default: 'planned']
  created_at datetime [not null]
  updated_at datetime [not null]  
}

// Stores multiple action steps for one SMART goal
Table goal_action_steps {
  step_id bigint [pk, increment]
  goal_id bigint [not null, ref: > smart_goals.goal_id, note: 'On delete: cascade']
  step_order int [not null]
  step_description text [not null]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (goal_id, step_order) [unique]
  }
}


// =====================================================
// NETWORKING MODULE
// =====================================================

// Stores upcoming networking events created by students
Table networking_events {
  event_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  event_name varchar(60) [not null]
  event_datetime datetime [not null]
  location varchar(100)
  details text
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores questions prepared for a networking event
Table networking_event_questions {
  question_id bigint [pk, increment]
  event_id bigint [not null, ref: > networking_events.event_id, note: 'On delete: cascade']
  question_order int [not null]
  question_text text [not null]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (event_id, question_order) [unique]
  }
}

// Stores industry contacts met by students
Table industry_contacts {
  contact_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  contact_name varchar(100) [not null]
  company varchar(100)
  progress_notes text [note: 'Follow-up notes and interaction history']
  date_met date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Supports multiple contact methods per contact instead of fixed columns
Table industry_contact_methods {
  method_id bigint [pk, increment]
  contact_id bigint [not null, ref: > industry_contacts.contact_id, note: 'On delete: cascade']
  method_type varchar(25)
  method_value varchar(255) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// =====================================================
// COMPETENCY TRACKING
// =====================================================

// High-level competency groups such as Category 1, 2, 3
Table competency_groups {
  group_id bigint [pk, increment]
  display_id varchar(20) [not null, unique]
  group_name varchar(100) [not null]
  description text
  discontinued_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Individual competency indicators under each group
Table competency_indicators {
  indicator_id bigint [pk, increment]
  group_id bigint [not null, ref: > competency_groups.group_id, note: "On delete: cascade"]
  display_id varchar(5) [not null, unique]
  indicator_name varchar(255) [not null]
  description text [not null]
  indicator_link varchar(255) [note: 'Reference link for this indicator']
  discontinued_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Student competency records
Table competency_entries {
  entry_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id]
  indicator_id bigint [not null, ref: > competency_indicators.indicator_id]

  // Reflection details
  experience_title varchar(255) [not null, note: 'Short title of the experience']
  associated_year int [not null, note: 'Academic year']
  experience_tasks text [not null, note: 'Description of experience and tasks']
  key_learnings text [note: 'Key learnings from that experience']
  future_applications text [note: 'How the learning will be apllied in future']

  entry_level_id bigint [not null, ref: > competency_entry_levels.entry_level_id, note: "On delete: restrict"]
  entry_status_id bigint [not null, ref: > competency_entry_statuses.entry_status_id, note: "On delete: restrict"]
  start_date date [not null]
  end_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores the level options that can be selected
Table competency_entry_levels {
  entry_level_id bigint [pk, increment]
  competency_level varchar(15) [not null, note: "Emerging, Developing, Proficient, Confident"]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores status options that can be applied to an entry
Table competency_entry_statuses {
  entry_status_id bigint [pk, increment]
  entry_status varchar(15) [not null, note: "Draft, Submitted, Reviewed"]
  created_at datetime [not null]
  updated_at datetime [not null]
}


// Stores multiple evidence each competency entry
Table competency_evidence {
  evidence_id bigint [pk, increment]
  entry_id bigint [not null, ref: > competency_entries.entry_id, note: 'On delete: cascade']
  evidence_type_id bigint [not null, ref: > competency_evidence_types.evidence_type_id, note: 'On delete: restrict']
  evidence_value varchar(500) [not null, note: 'Stores evidence path']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table competency_evidence_types {
  evidence_type_id bigint [pk, increment]
  evidence_type varchar(15) [not null, note: "url, file"]
  created_at datetime [not null]
  updated_at datetime [not null]  
}


// =====================================================
// STAFF FEEDBACK
// =====================================================

// Stores staff feedback for SMART goals
Table goal_feedback {
  feedback_id bigint [pk, increment]
  goal_id bigint [not null, ref: > smart_goals.goal_id, note: 'On delete: cascade']
  staff_id bigint [ref: > users.user_id]
  feedback_content text [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores staff feedback for competency entries
Table competency_feedback {
  feedback_id bigint [pk, increment]
  entry_id bigint [not null, ref: > competency_entries.entry_id, note: 'On delete: cascade']
  staff_id bigint [ref: > users.user_id]
  feedback_content text [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Keep this if each staff member should only manage assigned students
Table mentor_student_mapping {
  mapping_id bigint [pk, increment]
  staff_id bigint [not null, ref: > users.user_id, note: 'On delete: cascade']
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  assigned_at datetime [not null]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (staff_id, profile_id) [unique]
  }
}

Table system_audit_logs {
  log_id bigint [pk, increment]
  admin_id bigint [ref: > users.user_id, note: 'On delete: set null']
  action_type varchar(100) [not null]
  target_table varchar(100) [not null]

}
