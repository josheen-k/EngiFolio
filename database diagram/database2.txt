// =====================================================
// USER & ACCESS MANAGEMENT
// =====================================================

// Stores system roles such as Student, Staff, Admin
Table roles {
  role_id bigint [pk, increment]
  role_name varchar(50) [not null, unique, note: 'Student, Staff, Admin']
}

// Stores login and account information
Table users {
  user_id bigint [pk, increment]
  username varchar(100) [not null, unique, note: 'Student ID or Staff ID']
  email varchar(255) [not null, unique]
  password_hash varchar(255) [not null]
  role_id bigint [not null, ref: > roles.role_id]
  account_status varchar(25) [note: "CHECK (account_status IN ('active', 'disabled'))", not null, default: 'active']
  created_at datetime [not null]
  updated_at datetime [not null]
}


// =====================================================
// STUDENT PROFILE
// =====================================================

// Stores core student profile information
Table student_profiles {
  profile_id bigint [pk, increment]
  user_id bigint [not null, unique, ref: - users.user_id, note: 'On delete: cascade']
  first_name varchar(100)
  preferred_name varchar(100) [note: 'Optional display name']
  last_name varchar(100) [not null]
  degree_title varchar(255)
  specialization varchar(255)
  personal_intro text
  upcoming_actions text [note: 'Future personal development actions']
  profile_image_url varchar(500)
  created_at datetime [not null]
  updated_at datetime [not null]
}

// [NEW] Centralized link storage instead of mixing fixed URL columns and extra links
Table student_links {
  link_id bigint [pk, increment]
  profile_id bigint [not null, ref: > student_profiles.profile_id, note: 'On delete: cascade']
  link_type varchar(25) [note: "CHECK (link_type IN ('linkedin', 'resume', 'cover_letter', 'github', 'portfolio', 'other'))", not null]
  link_label varchar(100) [note: 'Display label shown to users']
  link_url varchar(500) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (profile_id, link_type, link_url) [unique]
  }
}

// Separated from student_profiles because elevator pitch belongs more naturally to networking
Table elevator_pitches {
  pitch_id bigint [pk, increment]
  user_id bigint [not null, unique, ref: - users.user_id, note: 'On delete: cascade']
  pitch_text text
  created_at datetime [not null]
  updated_at datetime [not null]
}

// =====================================================
// CAREER DEVELOPMENT PLAN (CDP)
// =====================================================

// Each student can have one CDP per year
Table career_development_plans {
  plan_id bigint [pk, increment]
  user_id bigint [not null, ref: > users.user_id, note: 'On delete: cascade']
  plan_year int [not null]
  professional_interests text
  employers_of_interest text
  networking_plan text
  personal_values text
  extracurriculars text
  development_focus text
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (user_id, plan_year) [unique]
  }
}


// =====================================================
// SMART GOALS
// =====================================================

// Stores SMART goals under a specific career plan
Table smart_goals {
  goal_id bigint [pk, increment]
  plan_id bigint [not null, ref: > career_development_plans.plan_id, note: 'On delete: cascade']
  goal_description text [not null]
  timeline text
  progress_notes text
  learnings text
  start_date date
  end_date date
  completion_date date
  completion_notes text
  status varchar(25) [note: "CHECK (status IN ('planned', 'in_progress', 'completed'))", not null, default: 'planned']
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores multiple action steps for one SMART goal
Table goal_action_steps {
  step_id bigint [pk, increment]
  goal_id bigint [not null, ref: > smart_goals.goal_id, note: 'On delete: cascade']
  step_order int [not null]
  step_description text [not null]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (goal_id, step_order) [unique]
  }
}


// =====================================================
// NETWORKING MODULE
// =====================================================

// Stores upcoming networking events created by students
Table networking_events {
  event_id bigint [pk, increment]
  user_id bigint [not null, ref: > users.user_id, note: 'On delete: cascade']
  event_name varchar(255) [not null]
  event_datetime datetime [not null]
  location varchar(255)
  details text
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Stores questions prepared for a networking event
Table networking_event_questions {
  question_id bigint [pk, increment]
  event_id bigint [not null, ref: > networking_events.event_id, note: 'On delete: cascade']
  question_order int [not null]
  question_text text [not null]
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (event_id, question_order) [unique]
  }
}

// Stores industry contacts met by students
Table industry_contacts {
  contact_id bigint [pk, increment]
  user_id bigint [not null, ref: > users.user_id, note: 'On delete: cascade']
  contact_name varchar(255) [not null]
  company varchar(255)
  progress_notes text [note: 'Follow-up notes and interaction history']
  date_met date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// [NEW] Supports multiple contact methods per contact instead of fixed columns
Table industry_contact_methods {
  method_id bigint [pk, increment]
  contact_id bigint [not null, ref: > industry_contacts.contact_id, note: 'On delete: cascade']
  method_type varchar(25) [note: "CHECK (contact_method IN ('phone', 'email', 'linkedin', 'website', 'other'))", not null]
  method_value varchar(500) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// =====================================================
// COMPETENCY TRACKING
// =====================================================

// High-level competency groups such as Category 1, 2, 3
Table competency_groups {
  group_id bigint [pk, increment]
  display_id varchar(20) [not null, unique]
  group_name varchar(255) [not null]
  description text
  discontinued_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Individual competency indicators under each group
Table competency_indicators {
  indicator_id bigint [pk, increment]
  group_id bigint [not null, ref: > competency_groups.group_id]
  display_id varchar(20) [not null, unique]
  indicator_name varchar(255) [not null]
  description text [not null]
  indicator_link varchar(500) [note: 'Reference link for this indicator']
  discontinued_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Student competency records
Table competency_entries {
  entry_id bigint [pk, increment]
  user_id bigint [not null, ref: > users.user_id]
  indicator_id bigint [not null, ref: > competency_indicators.indicator_id]

  // Reflection details
  experience_title varchar(255) [not null, note: 'Short title of the experience']
  associated_year int [not null, note: 'Academic year']
  experience_tasks text [not null, note: 'Description of experience and tasks']
  key_learnings text [note: 'Key learnings from that experience']
  future_applications text [note: 'How the learning will be apllied in future']

  level varchar(25) [note: "CHECK (level IN ('Emerging', 'Developing', 'Proficient', 'Competent'))", not null]
  status varchar(25) [note: "CHECK (status IN ('Draft', 'Submitted', 'Reviewed'))", not null, default: 'draft']
  start_date date [not null]
  end_date date
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (user_id, indicator_id, status)
  }
}

// Stores multiple evidence each competency entry
Table competency_evidence {
  evidence_id bigint [pk, increment]
  entry_id bigint [not null, ref: > competency_entries.entry_id, note: 'On delete: cascade']
  evidence_type varchar(25) [note: "CHECK (evidence_type IN ('url', 'file'))", not null]
  evidence_value varchar(500) [not null, note: 'Stores evidence path']
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (entry_id)
  }
}

// =====================================================
// CDL MODULES
// =====================================================

// Stores available learning modules
Table cdl_modules {
  cdl_id bigint [pk, increment]
  title varchar(255) [not null]
  description text
  module_url varchar(500)
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Tracks student progress in each CDL module
Table student_cdl_progress {
  progress_id bigint [pk, increment]
  user_id bigint [not null, ref: > users.user_id, note: 'On delete: cascade']
  cdl_id bigint [not null, ref: > cdl_modules.cdl_id, note: 'On delete: cascade']
  status varchar(25) [note: "CHECK (status IN ('Not Started', 'In Progress', 'Completed''))", not null, default: 'not_started']
  progress_percentage int [not null, default: 0, note: 'Should be validated as 0-100 in application logic']
  last_accessed_at datetime
  completed_at datetime
  created_at datetime [not null]
  updated_at datetime [not null]

  indexes {
    (user_id, cdl_id) [unique]
  }
}
