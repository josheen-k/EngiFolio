// --- 1. Identity & Access Management (IAM) ---
Table users {
  user_id int [pk, increment]
  username varchar [unique, note: 'Student ID or Staff ID']
  password_hash varchar
  full_name varchar
  email varchar [unique]
  role_id int [note: '1-Student, 2-Teaching Staff, 3-IT Staff']
  account_status enum('active', 'disabled') [default: 'active']
  created_at timestamp [default: `now()`]
}

// --- 2. Student Portfolio & Career Planning ---
Table student_profiles {
  profile_id int [pk, increment]
  user_id int [unique, ref: - users.user_id]
  degree_title varchar
  commencement_date date [note: 'Critical for tracking part-time duration']
  bio_intro text
  elevator_pitch text
  resume_url varchar
  linkedin_url varchar
  job_application_notes text [note: 'Stores selection criteria responses']
  updated_at timestamp
}

Table networking_questions {
  question_id int [pk, increment]
  profile_id int [ref: > student_profiles.profile_id]
  question_text text
  intended_audience varchar [note: 'e.g., Industry Mentor, Recruiter']
}

Table career_development_plans {
  plan_id int [pk, increment]
  user_id int [ref: > users.user_id]
  calendar_year int [note: 'Groups plans by year instead of academic level']
  professional_interests text
  target_employers text
  personal_values text
  extracurricular_activities text
}

Table smart_goals {
  goal_id int [pk, increment]
  plan_id int [ref: > career_development_plans.plan_id]
  goal_description text
  action_steps text
  timeline varchar
  progress_notes text
  key_learnings text
  status enum('planned', 'in_progress', 'completed')
}

// --- 3. EA Competency Tracking ---
Table ea_competency_standards {
  comp_id int [pk, increment]
  comp_code varchar [unique, note: 'e.g., 1.1, 2.3, 3.6']
  category enum('Knowledge and Skill Base', 'Engineering Application Ability', 'Professional and Personal Attributes')
  title varchar
  description text
}

Table attainment_levels {
  level_id int [pk, increment]
  level_name enum('Emerging', 'Developing', 'Proficient', 'Competent')
  description text
}

Table competency_evaluations {
  evaluation_id int [pk, increment]
  student_id int [ref: > users.user_id]
  comp_id int [ref: > ea_competency_standards.comp_id]
  level_id int [ref: > attainment_levels.level_id]
  experience_date date [note: 'The actual date the activity occurred']
  context_category enum('Coursework', 'Internship', 'Volunteering', 'Club/Society')
  reflection_text text [note: 'Description, Learning, and Future Plan']
  status enum('draft', 'submitted', 'reviewed')
  last_modified_by int [ref: > users.user_id, note: 'Allows tracking of IT Staff overrides']
  updated_at timestamp
}

Table evaluation_evidence {
  evidence_id int [pk, increment]
  evaluation_id int [ref: > competency_evaluations.evaluation_id]
  evidence_type varchar [note: 'Document, Video, URL']
  file_url varchar
  description text
}

// --- 4. Staff Feedback & Supervision ---
Table mentor_student_mapping {
  mapping_id int [pk, increment]
  staff_id int [ref: > users.user_id]
  student_id int [ref: > users.user_id]
  assigned_at timestamp [default: `now()`]
}

Table staff_feedback {
  feedback_id int [pk, increment]
  evaluation_id int [ref: > competency_evaluations.evaluation_id]
  staff_id int [ref: > users.user_id]
  feedback_content text
  is_verified boolean [default: false]
  last_modified_by int [ref: > users.user_id]
  created_at timestamp [default: `now()`]
}

// --- 5. System Administration & Auditing ---
Table system_audit_logs {
  log_id bigint [pk, increment]
  admin_id int [ref: > users.user_id]
  action_type varchar [note: 'MODIFY_USER, OVERRIDE_EVALUATION']
  target_table varchar
  target_row_id int
  previous_value_snapshot json [note: 'Crucial for IT Staff accountability']
  new_value_snapshot json
  timestamp timestamp [default: `now()`]
}
