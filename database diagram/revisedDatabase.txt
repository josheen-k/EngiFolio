// EngiFolio database for EA Competencies

// Student table linked to the uni
Table student {
  student_id bigint [pk, not null, unique]
  first_name varchar(100) [not null]
  last_name varchar(100) [not null]
  program varchar(255) [not null]
}

Table student_profile {
  profile_id bigint [pk, increment]
  student_id bigint [not null, unique]
  introduction varchar(255)  [note: 'Write a brief introduction about yourself (e.g. your degree, areas of interest in engineering, and key strengths).']
  elevator_pitch text  [note: 'Prepare a 30-second pitch to introduce yourself at networking events or interviews.']
  resume_url varchar(500)  [note: 'Insert a link to your most recent resume (e.g. via cloud storage or portfolio).']
  linked_in_url varchar(500)  [note: 'Include a link to your LinkedIn or other professional profile (if available).'] 
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table career_development_plan {
  development_plan_id bigint [pk, increment]
  student_id bigint [not null, unique]
  career_goal varchar(500) [note: 'Career Goal on Graduation']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table ea_competency_development {
  ea_competency_development_id bigint [pk, increment]
  student_id bigint [not null, unique]
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Table linked to EA competency development
Table ea_competency {
  ea_competency_id bigint [pk, increment]
  ea_competency_development_id bigint
  ea_competency_code bigint [not null, unique]
  ea_category varchar(500)
  entering_level varchar(500) [note: 'Emerging, Developing, Proficient, Confident']
  skills_prior varchar(500) [note: 'Prior to Entering the Program - Skills Review']
  evidence_link varchar(500) [note: 'Include a link to relevant evidence of your experience and the skills you developed. This could include a report, presentation slides, drawings, designs, feedback from teachers, peer feedback, supervisor feedback from a part-time job, or evidence of activities you’ve led or organised through student societies, clubs, or volunteering etc.']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table ea_competency_yearly {
  ea_competency_yearly_id bigint [pk, increment]
  ea_competency_id bigint
  year_number bigint
  skill_level varchar(500) [note: 'Emerging, Developing, Proficient, Confident']
  skills_review varchar(500)
  evidence_link varchar(500) [note: 'Include a link to relevant evidence of your experience and the skills you developed. This could include a report, presentation slides, drawings, designs, feedback from teachers, peer feedback, supervisor feedback from a part-time job, or evidence of activities you’ve led or organised through student societies, clubs, or volunteering etc.']
}
