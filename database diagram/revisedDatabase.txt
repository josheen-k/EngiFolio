// EngiFolio database for EA Competencies

// Student table linked to the uni
Table student {
  student_id varchar(20) [pk]
  first_name varchar(100) [not null]
  last_name varchar(100) [not null]
  email varchar(255) [unique, not null]
  program varchar(255) [not null]
}

Table staff {
  staff_id varchar(20) [pk]
  first_name varchar(100) [not null]
  last_name varchar(100) [not null]
  email varchar(255) [unique, not null]
}

Table staff_profile {
  staff_profile_id bigint [pk]
  staff_id varchar(20) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table assigned_student {
  assigned_student_id varchar(20) [pk]
  staff_id varchar(20) [not null]
  student_id varchar(20) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
} 

Table student_profile {
  profile_id bigint [pk, increment]
  student_id  varchar(20) [not null, unique]
  introduction text  [note: 'Write a brief introduction about yourself (e.g. your degree, areas of interest in engineering, and key strengths).']
  elevator_pitch text  [note: 'Prepare a 30-second pitch to introduce yourself at networking events or interviews.']
  resume_url varchar(500)  [note: 'Insert a link to your most recent resume (e.g. via cloud storage or portfolio).']
  linked_in_url varchar(500)  [note: 'Include a link to your LinkedIn or other professional profile (if available).'] 
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table career_development_plan {
  development_plan_id bigint [pk, increment]
  student_id  varchar(20) [not null, unique]
  career_goal varchar(500) [note: 'Career Goal on Graduation']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table ea_competency_development {
  ea_competency_development_id bigint [pk, increment]
  student_id  varchar(20) [not null, unique]
  created_at datetime [not null]
  updated_at datetime [not null]
}


// Tables linked directly to the introduction section of the student engineering profile
Table event {
    event_id bigint [pk, increment]
    profile_id bigint [not null]
    name char(255) [not null]
    event_date date [not null]
    event_time time [not null]
    location char(255)
    details text
    created_at datetime [not null]
    updated_at datetime [not null]
}

Table networking_questions {
    networking_questions_id bigint [pk, increment]
    event_id bigint [not null]
    networking_questions text [note: 'List 3–5 thoughtful questions you can ask employers or professionals to start meaningful conversations.']
    created_at datetime [not null]
    updated_at datetime [not null]
}

Table industry_contact {
    industry_contact_id bigint [pk, increment]
    profile_id bigint [not null]
    name char(255) [not null]
    company char(255) [not null]
    contact_info text
    notes text
    date_met date
    created_at datetime [not null]
    updated_at datetime [not null]
}

Table job_resources {
  job_resources_id bigint [pk, increment]
  profile_id bigint [not null]
  title varchar(255) [not null]
  body text [note: 'Store responses to selection criteria, interview practice videos, or other supporting materials for job applications.']
  file_path varchar(500) [note: 'if applicable']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table achievement_cert {
  achievement_cert_id bigint [pk, increment]
  profile_id bigint [not null]
  title varchar(255) [not null]
  body text [note: 'Upload awards, scholarships, or other certificates that demonstrate your academic or extracurricular success.']
  file_path varchar(500) [note: 'if applicable']
  issued_date date
  expiry_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table attainment_cert {
  attainment_cert_id bigint [pk, increment]
  profile_id bigint [not null]
  title varchar(255) [not null]
  body text [note: 'Include training credentials such as White Card, First Aid, WHS training, or other job-related certifications.']
  file_path varchar(500) [note: 'if applicable']
  issued_date date
  expiry_date date
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table portfolio_note {
  portfolio_note_id bigint [pk, increment]
  profile_id bigint [not null]
  title varchar(255) [not null]
  body text [note: 'Add links to relevant portfolio items, project work, or documentation of achievements.']
  file_path varchar(500) [note: 'if applicable']
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Tables linked to career development plan

Table networking_plan {

  career_planning_overview_id bigint [not null]

Table smart_goal {
  smart_goal_id bigint [pk, increment]
  development_plan_id bigint [not null]
  timeline varchar(500) [note: 'Career Goal on Graduation']
  progress_notes text [note: 'Career Goal on Graduation']
  key_learnings text [note: 'Career Goal on Graduation']
  start_date datetime [not null]
  end_date datetime [not null]
  completion_date datetime
  completion_notes text
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table smart_goal_feedback {
  smart_feedback_id bigint [pk, increment]
  smart_goal_id bigint [not null]
  staff_id bigint [not null]
  feedback text
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Individual action steps to take to achieve the smart goal
Table action_step {
  action_step_id bigint [pk, increment]
  smart_goal_id bigint [not null]
  description varchar(500) [note: 'Career Goal on Graduation']
  completed tinyint(1) [note: 'Yes or no']
  created_at datetime [not null]
  updated_at datetime [not null]
}

// Table linked to EA competency development
Table ea_competency {
  ea_competency_id bigint [pk, increment]
  ea_competency_development_id bigint [not null]
  ea_competency_code varchar(20) [not null]
  ea_category varchar(255) [not null]
  ea_competency varchar(500) [not null]
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table ea_competency_yearly {
  ea_competency_yearly_id bigint [pk, increment]
  ea_competency_id bigint [not null]
  year_number int [not null, note: 'Sequence numbers beginning at 0 (Entering)']
  skill_level varchar(50) [not null, note: 'Emerging, Developing, Proficient, Confident']
  skills_review text
  evidence_link varchar(500) [note: 'Include a link to relevant evidence of your experience and the skills you developed. This could include a report, presentation slides, drawings, designs, feedback from teachers, peer feedback, supervisor feedback from a part-time job, or evidence of activities you’ve led or organised through student societies, clubs, or volunteering etc.']
  created_at datetime [not null]
  updated_at datetime [not null]
}

Table competency_feedback {
  competency_feedback_id bigint [pk, increment]
  ea_competency_yearly_id bigint [not null]
  staff_id bigint [not null]
  feedback text
  created_at datetime [not null]
  updated_at datetime [not null]
}
